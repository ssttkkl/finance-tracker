230|```python
231|"""建行储蓄卡 XLS 转换器"""
232|import xlrd
233|import re
234|from datetime import datetime, timedelta
235|
236|
237|def read_ccb_debit(path: str) -> list[dict]:
238|    """解析建行个人活期账户交易明细 XLS，返回 rec dict 列表"""
239|    wb = xlrd.open_workbook(path)
240|    sh = wb.sheet_by_index(0)
241|
242|    # 提取卡号（第 2 行，row index 1）
243|    card = ""
244|    for c in range(sh.ncols):
245|        val = str(sh.cell_value(1, c))
246|        m = re.search(r"卡号/账号[：:](\d+)", val)
247|        if m:
248|            card = m.group(1)
249|            break
250|    card_last4 = card[-4:] if card else ""
251|
252|    # 币别映射
253|    cur_map = {"人民币元": "CNY", "人民币": "CNY", "美元": "USD", "港币": "HKD", "日元": "JPY"}
254|
255|    records = []
256|
257|    # 数据从第 5 行开始（row index 4），是 Excel 的第 5 行（前面是标题/卡号/合计/表头）
258|    for ri in range(4, sh.nrows):
259|        row_vals = [sh.cell_value(ri, c) for c in range(sh.ncols)]
260|
261|        # 跳过空行
262|        if not row_vals or not row_vals[0]:
263|            continue
264|
265|        # 确定数据起始：第 1 列是序号（纯数字字符串如 '1'/'2'）
266|        seq = str(int(float(str(row_vals[0])))) if row_vals[0] else ""
267|        if not seq:
268|            continue
269|
270|        summary = str(row_vals[1] or "").strip()
271|        cur_str = str(row_vals[2] or "").strip()
272|        currency = cur_map.get(cur_str, "CNY")
273|
274|        # 日期 YYYYMMDD → YYYY-MM-DD
275|        date_raw = str(int(float(str(row_vals[4])))) if row_vals[4] else ""
276|        date = f"{date_raw[:4]}-{date_raw[4:6]}-{date_raw[6:8]} 00:00:00" if len(date_raw) == 8 else ""
277|
278|        # 金额
279|        amt_str = str(row_vals[5] or "").replace(",", "").strip()
280|        try:
281|            amount = round(float(amt_str), 2)
282|        except ValueError:
283|            continue
284|
285|        # 对方账号与户名：取 / 后部分
286|        acct_name = str(row_vals[8] or "").strip() if len(row_vals) > 8 else ""
287|        if "/" in acct_name:
288|            counterparty = acct_name.split("/", 1)[1].strip()
289|        else:
290|            counterparty = acct_name
291|
292|        category = "expense" if amount < 0 else "income"
293|
294|        records.append({
295|            "date": date,
296|            "amount": amount,
297|            "currency": currency,
298|            "card_number": card_last4,
299|            "counterparty": counterparty,
300|            "description": summary,
301|            "category": category,
302|            "payment_method": f"建行储蓄卡({card_last4})",
303|            "platform": "",  # do_convert 中会调 _infer_platform
304|        })
305|
306|    # 退款配对
307|    records = _pair_ccb_refunds(records)
308|
309|    wb.release_resources()
310|    return records
311|
312|
313|def _pair_ccb_refunds(records: list) -> list:
314|    """建行储蓄卡退款配对核销
315|
316|    匹配条件：
317|    1. 金额精确相等
318|    2. 退款在消费日后 30 天内
319|    3. 附加信号：脱敏名末位字符匹配（优先级加权）
320|    """
321|    expenses = [(i, r) for i, r in enumerate(records) if r["category"] == "expense"]
322|    refunds = [(i, r) for i, r in enumerate(records) if r["category"] == "income" and "退款" in r.get("description", "")]
323|
324|    paired_exp = set()
325|    paired_ref = set()
326|
327|    for ri, ref in refunds:
328|        ref_amt = abs(ref["amount"])
329|        ref_date = ref["date"]
330|        ref_cpy = ref["counterparty"]
331|        ref_last = ref_cpy[-1] if ref_cpy else ""
332|
333|        candidates = []
334|        for ei, exp in expenses:
335|            if ei in paired_exp:
336|                continue
337|            if abs(abs(exp["amount"]) - ref_amt) > 0.005:
338|                continue
339|            if exp["date"] > ref_date:
340|                continue
341|            # 30 天窗口
342|            try:
343|                exp_dt = datetime.strptime(exp["date"][:10], "%Y-%m-%d")
344|                ref_dt = datetime.strptime(ref_date[:10], "%Y-%m-%d")
345|                if (ref_dt - exp_dt).days > 30:
346|                    continue
347|            except ValueError:
348|                continue
349|
350|            # 附加信号：末位字符匹配（排除「微信转账」等平台退款）
351|            bonus = 0
352|            exp_cpy = exp["counterparty"]
353|            exp_last = exp_cpy[-1] if exp_cpy else ""
354|            if ref_last and exp_last and ref_last == exp_last and ref_cpy not in ("微信转账",):
355|                bonus = 1
356|
357|            candidates.append((ei, bonus, exp["date"]))
358|
359|        if not candidates:
360|            # 尝试纯金额+日期匹配（脱敏名末位不一致的兜底）
361|            for ei, exp in expenses:
362|                if ei in paired_exp:
363|                    continue
364|                if abs(abs(exp["amount"]) - ref_amt) > 0.005:
365|                    continue
366|                if exp["date"] > ref_date:
367|                    continue
368|                try:
369|                    exp_dt = datetime.strptime(exp["date"][:10], "%Y-%m-%d")
370|                    ref_dt = datetime.strptime(ref_date[:10], "%Y-%m-%d")
371|                    if (ref_dt - exp_dt).days > 30:
372|                        continue
373|                    candidates.append((ei, 0, exp["date"]))
374|                except ValueError:
375|                    continue
376|
377|        if candidates:
378|            # 优先末位匹配，其次最近消费
379|            candidates.sort(key=lambda c: (c[1], c[2]), reverse=True)
380|            best = candidates[0][0]
381|
382|            exp_amt = abs(expenses[[e[0] for e in expenses].index(best)][1]["amount"])
383|            if abs(exp_amt - ref_amt) < 0.005:
384|                # 全额退款 → 删除两条
385|                paired_exp.add(best)
386|                paired_ref.add(ri)
387|            else:
388|                # 部分退款 → 调整净额，删除退款
389|                target = expenses[[e[0] for e in expenses].index(best)]
390|                target_idx = target[0]
391|                records[target_idx]["amount"] = round(records[target_idx]["amount"] + ref_amt, 2)
392|                paired_ref.add(ri)
393|
394|    # 构建结果：排除已配对的退款和已全额配对的消费
395|    result = []
396|    for i, rec in enumerate(records):
397|        if i in paired_ref:
398|            continue
399|        if i in paired_exp:
400|            continue
401|        result.append(rec)
402|
403|    return result
404|```
